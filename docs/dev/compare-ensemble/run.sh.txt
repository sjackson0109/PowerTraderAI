#!/usr/bin/env bash
# The ensemble and sizing exploration, in a scratch sandbox copied from the nine-coin
# comparison's: no real user folders, no keys (the keyring fails), its Settings and cached
# public candles. Code: the combined tree wt-all6 (main + #267-#287 + tonight's branches).
set -u
S="<the scratch folder>"
B="$S/cmpens/sbx"
PY="$S/venv2/Scripts/python.exe"
APP="$S/wt-all6/app"
DRIVER="$S/cmpens/run_compare_ensemble.py"
C9="$S/wt-all6/docs/dev/run_compare9.py"
export POWERTRADER_HOME="$(cygpath -w "$B/home")"
export APPDATA="$(cygpath -w "$B/roaming")"
export LOCALAPPDATA="$(cygpath -w "$B/local")"
export PYTHON_KEYRING_BACKEND=keyring.backends.fail.Keyring PYTHONNOUSERSITE=1
export PYTHONDONTWRITEBYTECODE=1 PYTHONIOENCODING=utf-8 POWERTRADER_USER_REGION=GLOBAL
cd "$B" || exit 1
"$PY" -u "$(cygpath -w "$DRIVER")" "$(cygpath -w "$APP")" "$(cygpath -w "$S/cmpens/out")" "$(cygpath -w "$C9")"
