#!/usr/bin/env bash
# FDS-MH-001 phase E, batch 1's walk-forward, in a scratch sandbox: no real user folders,
# no keys (the keyring fails), Settings from the shipped template (Coinbase primary),
# candles from the cache PV-1 read (copied in, read offline).
# usage: run.sh freeze | run <batch id>
set -u
S="<the scratch folder>"
B="$S/mh-e/sbx"
PY="$S/venv/Scripts/python.exe"
APP="<this repo>/app"
DRIVER="<this repo>/docs/dev/run_mh_batch1.py"
export POWERTRADER_HOME="$(cygpath -w "$B/home")"
export APPDATA="$(cygpath -w "$B/roaming")"
export LOCALAPPDATA="$(cygpath -w "$B/local")"
export PYTHON_KEYRING_BACKEND=keyring.backends.fail.Keyring PYTHONNOUSERSITE=1
export PYTHONDONTWRITEBYTECODE=1 PYTHONIOENCODING=utf-8 POWERTRADER_USER_REGION=GLOBAL
cd "$B" || exit 1
"$PY" -u "$(cygpath -w "$DRIVER")" "$(cygpath -w "$APP")" "$@"
